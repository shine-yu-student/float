```c++
// #define Redshift_Debug
#ifdef Redshift_Debug
#define debug(...) fprintf(stderr, __VA_ARGS__)
#include <chrono>
#else
#define debug(...)
#endif
#include <iostream>
#include <utility>
#include <vector>

using namespace std;
const int N = 5e5 + 10;
int n, vis[N << 1], vidx, q[N << 1], lb, qidx; // 0->pos, 1->neg
using pii = pair<int, int>;
vector<int> road[N];
vector<int> req[N << 1];
bool vld_1;
struct st
{
	int x, y, id;
	int inn_id(int tx)
	{
		return tx == y ? id : id + n;
	}
	int out_id(int tx)
	{
		return tx == x ? id : id + n;
	}
};
st ed[N];
void opr()
{
	static int ti = 0;
	while (lb < qidx)
	{
		lb++;
		ti = q[lb];
		for (auto &i : req[ti])
		{
			if (vis[i] == vidx)
				continue;
			q[++qidx] = i;
			vis[i] = vidx;
		}
	}
}
void init_global()
{
	cin.tie(0);
	cout.tie(0);
	ios::sync_with_stdio(0);
}
void init_local()
{
	lb = qidx = 0;
	vld_1 = true;
	cin >> n;
	for (int i = 1; i <= n; i++)
		road[i].clear();
	for (int i = 1; i <= (n << 1); i++)
		req[i].clear();
	for (int i = 1; i < n; i++)
	{
		cin >> ed[i].x >> ed[i].y;
		road[ed[i].x].emplace_back(i);
		road[ed[i].y].emplace_back(i);
		ed[i].id = i;
	}
	for (int i = 1, mx1, mx2; i <= n; i++)
	{
		if (road[i].size() == 1)
			continue;
		mx1 = mx2 = 0;
		for (auto &j : road[i])
		{
			if (mx1 < j)
				mx2 = mx1, mx1 = j;
			else if (mx2 < j)
				mx2 = j;
		}
		// if mx1 in, then mx2 must in
		req[ed[mx1].inn_id(i)].emplace_back(ed[mx2].inn_id(i));
		// if mx2 out, then mx1 must out
		req[ed[mx2].out_id(i)].emplace_back(ed[mx1].out_id(i));
	}
}
void run()
{
	vidx++;
	for (int i = 1; i <= n; i++)
	{
		if (road[i].size() != 1)
			continue;
		q[++qidx] = ed[road[i][0]].out_id(i);
		vis[q[qidx]] = vidx;
	}
	opr();
	for (int i = 1; i < n; i++)
	{
		if (vis[i] == vidx or vis[i + n] == vidx)
			continue;
		q[++qidx] = i;
		vis[i] = vidx;
		opr();
	}
	for (int i = 1; i < n; i++)
	{
		if (vis[i] == vidx and vis[i + n] == vidx)
		{
			vld_1 = false;
			break;
		}
	}
	if (!vld_1)
	{
		lb = qidx = 0;
		q[++qidx] = 1;
		vidx++;
		for (int i = 1, ti, nxt; i <= qidx; i++)
		{
			ti = q[i];
			for (auto &j : road[ti])
			{
				nxt = ed[j].x ^ ed[j].y ^ ti;
				if (vis[j] == vidx or vis[j + n] == vidx)
					continue;
				vis[ed[j].out_id(ti)] = vidx;
				q[++qidx] = nxt;
			}
		}
	}
	cout << (vld_1 ? 1 : 2) << '\n';
	for (int i = 1; i < n; i++)
	{
		cout << (vis[i] == vidx ? ed[i].y : ed[i].x) << " \n"[i == n - 1];
	}
}
int main()
{
#ifdef Redshift_Debug
	auto st = chrono::high_resolution_clock::now();
#endif
	int T = 1;
	init_global();
	cin >> T;
	while (T--)
	{
		init_local();
		run();
	}
#ifdef Redshift_Debug
	auto ed = chrono::high_resolution_clock::now();
	fprintf(stderr, "%.9lf\n", (ed - st).count() / 1e9);
#endif
}
```
